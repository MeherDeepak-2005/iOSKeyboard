cmake_minimum_required(VERSION 3.22)
project(KBMarisaExperiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(KB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. decoder-core)
file(GLOB_RECURSE MARISA_SOURCES CONFIGURE_DEPENDS vendor/lib/*.cc)
add_library(kb-marisa STATIC ${MARISA_SOURCES})
target_include_directories(kb-marisa PUBLIC vendor/include PRIVATE vendor/lib)
add_library(marisa-lexicon STATIC MarisaLexicon.cpp)
target_link_libraries(marisa-lexicon PUBLIC DecoderCore kb-marisa)
target_include_directories(marisa-lexicon PUBLIC .)
target_compile_options(marisa-lexicon PRIVATE -Wall -Wextra -Werror)
add_executable(marisa-experiment Experiment.cpp)
target_link_libraries(marisa-experiment PRIVATE marisa-lexicon)
