cmake_minimum_required(VERSION 3.22)
project(XploreeAndroidDecoder LANGUAGES CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "Use the Android NDK toolchain for this adapter; host tests use native/CMakeLists.txt")
endif()
set(KB_BUILD_TESTS OFF CACHE BOOL "No host test executable in the Android library" FORCE)
add_subdirectory(.. "${CMAKE_CURRENT_BINARY_DIR}/decoder-core")
add_library(xploree_decoder SHARED DecoderJni.cpp)
target_link_libraries(xploree_decoder PRIVATE DecoderCore)
target_compile_features(xploree_decoder PRIVATE cxx_std_20)
target_compile_options(xploree_decoder PRIVATE -Wall -Wextra -Werror)
